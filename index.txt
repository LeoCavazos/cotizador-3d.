<script>
  const GITHUB_TOKEN = "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";

  function quickSend(text) {
    document.getElementById('userInput').value = text;
    sendMessage();
  }

  async function sendMessage() {
    const input = document.getElementById('userInput');
    const text = input.value.trim();
    if (!text) return;

    appendMessage(text, 'user');
    input.value = '';

    const loadingId = appendMessage('Calculando volumen y tiempo estimado...', 'bot');

    try {
      const response = await fetch('https://models.inference.ai.azure.com/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${GITHUB_TOKEN}`
        },
        body: JSON.stringify({
          model: 'gpt-4o-mini',
          messages: [
            {
              role: 'system',
              content: 'Eres un experto en cotizaciones de impresión 3D (FDM y Resina SLA). Ofreces costos estimados claros en MXN detallando material (PLA, PETG, Resina), tiempo de impresión aproximado y recomendaciones técnicas según la pieza. Sé claro, profesional y usa emojis para estructurar la respuesta.'
            },
            { role: 'user', content: text }
          ]
        })
      });

      const data = await response.json();
      document.getElementById(loadingId).remove();

      if (data.choices && data.choices[0]) {
        appendMessage(data.choices[0].message.content, 'bot');
      } else {
        appendMessage("📐 **Cotización Estándar PLA:**
Para una pieza de esas características el costo promedio ronda entre $180 y $240 MXN con entrega en 24-48 horas.", 'bot');
      }
    } catch (err) {
      document.getElementById(loadingId).remove();
      appendMessage("📐 **Estimación rápida:** Pieza mediana en PLA ~ $220 MXN. Para obtener la cotización exacta requerimos el archivo .STL.", 'bot');
    }
  }

  function appendMessage(text, sender) {
    const chatBox = document.getElementById('chatBox');
    const msgDiv = document.createElement('div');
    const id = 'msg-' + Math.random().toString(36).substr(2, 9);
    msgDiv.id = id;
    msgDiv.className = `msg ${sender}`;
    msgDiv.style.whiteSpace = "pre-line";
    msgDiv.innerText = text;
    chatBox.appendChild(msgDiv);
    chatBox.scrollTop = chatBox.scrollHeight;
    return id;
  }

  let recognition;
  if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();
    recognition.lang = 'es-MX';
    recognition.onresult = (event) => {
      document.getElementById('userInput').value = event.results[0][0].transcript;
      document.getElementById('voiceBtn').classList.remove('recording');
    };
    recognition.onerror = () => document.getElementById('voiceBtn').classList.remove('recording');
  }

  function toggleVoice() {
    if (!recognition) return alert('Tu navegador no soporta reconocimiento de voz.');
    const btn = document.getElementById('voiceBtn');
    btn.classList.add('recording');
    recognition.start();
  }
